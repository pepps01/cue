name: Deploy Laravel to EC2

on:
  push:
    branches: [ "main" ]

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v3

      - name: Configure AWS Credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ secrets.AWS_REGION }}

      - name: Login to Amazon ECR
        id: login-ecr
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build, Tag, Push image
        env:
          ECR_REGISTRY: ${{ steps.login-ecr.outputs.registry }}
          ECR_REPOSITORY: ${{ secrets.ECR_REPOSITORY }}
          IMAGE_TAG: latest
        run: |
          docker build -t $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG .
          docker push $ECR_REGISTRY/$ECR_REPOSITORY:$IMAGE_TAG

      - name: Deploy to EC2
        uses: appleboy/ssh-action@v1.0.3
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ${{ secrets.EC2_USER }}
          key: ${{ secrets.EC2_SSH_KEY }}
          script: |
            docker pull ${{ secrets.AWS_ACCOUNT_ID }}.dkr.ecr.${{ secrets.AWS_REGION }}.amazonaws.com/${{ secrets.ECR_REPOSITORY }}:latest

            cd /home/${{ secrets.EC2_USER }}/app

            docker compose down
            docker compose up -d --remove-orphans
            docker-compose exec -T app php artisan migrate --force
            docker-compose exec -T app php artisan config:cache

name: Deploy Laravel App

on:
  push:
    branches: [ main ]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Deploy to EC2
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.HOST_NAME }}
          username: ${{ secrets.USERNAME }}
          key: ${{ secrets.CUE_SSH_KEY }}
          script: |
            # 1. Define where the app should live
            APP_DIR="/home/ubuntu/cue"
            REPO_URL="https://github.com/pepps01/cue.git"

            # 2. Smart Initial Setup
            # If the folder doesn't exist OR isn't a git repo, clone it fresh
            if [ ! -d "$APP_DIR/.git" ]; then
              echo "Re-initializing project folder..."
              sudo rm -rf "$APP_DIR"
              git clone $REPO_URL $APP_DIR
            fi

            # 3. Enter the directory
            cd $APP_DIR

            # 4. Update the code
            git pull origin main

            # 5. Build and Restart Containers
            # Checks if docker-compose.yml is there to avoid the "file not found" error
            if [ -f "docker-compose.yml" ]; then
              docker compose up -d --build
              
              # 6. Run Laravel Maintenance
              docker compose exec -T app php artisan migrate --force
              docker compose exec -T app php artisan config:cache
              docker compose exec -T app php artisan route:cache
            else
              echo "Error: docker-compose.yml not found!"
              exit 1
            fi
